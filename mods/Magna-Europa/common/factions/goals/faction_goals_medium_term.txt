		}
		any_collection_element = {

			}
			distance_to = {
				target = 159   # 1.19-migration: was vanilla 100 (Iceland)
				value < 2000
			}
		}
	}
	available = {

	}
	completed = {
		159 = { is_controlled_by_ROOT_or_ally = yes }   # 1.19-migration: was vanilla 100 (Iceland)
	}
	complete_effect = {
		add_faction_power_projection = 100
		custom_effect_tooltip = generic_skip_one_line_tt
		custom_effect_tooltip = TARGET_STATE_CONTROLLERS
		every_faction_member = { 
			visible_when_empty = yes
			limit = {
				controls_state = 159   # 1.19-migration: was vanilla 100 (Iceland)
			}
			add_faction_initiative = 1
		}
		159 = {   # 1.19-migration: was vanilla 100 (Iceland)
			add_building_construction = {
				type = anti_air_building
				level = 1
				instant_build = yes
			}
			add_building_construction = {
				type = air_base
				level = 1
				instant_build = yes
			}
			add_building_construction = {
				type = naval_base
				level = 1
				instant_build = yes
				province = {
					all_provinces = yes
					limit_to_naval_base = yes
				}
			}
		}
	}
	ai_will_do = {
		base = 1

			factor = 2 # Target of opportunity - already at war anyway

				owner = {
					has_war_with = ROOT
				}
			}
		}
		modifier = {
			add = 2 # If convoys being threatened try to secure the transatlantic supply lines better
			convoy_threat > 0.5
			collection_size = {
				input = {
					input = game:all_states
					operators = {
						limit = {
							is_controlled_by_ROOT_or_ally = yes
							OR = {
								is_on_continent = north_america
								is_on_continent = south_america
							}
						}

			}

					input = game:all_states

							is_controlled_by_ROOT_or_ally = yes
							OR = {
								is_on_continent = europe
								is_on_continent = africa
							}
						}
					}
					name = COLLECTION_FACTION_STATES_IN_EUROPE
				}
