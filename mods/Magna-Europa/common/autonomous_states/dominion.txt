autonomy_state = {
	id = autonomy_dominion
	
	use_overlord_color = yes   # 1.19-migration: mod keeps overlord color on all puppet tiers

	min_freedom_level = 0.8
	
	manpower_influence = 0.2
	
	rule = {
		desc = "RULE_DESC_IS_A_SUBJECT"
	
		can_not_declare_war = yes
		can_decline_call_to_war = no
		can_be_spymaster = yes
	}
	
	modifier = {
		autonomy_manpower_share = 0.0
		extra_trade_to_overlord_factor = 0.25
		overlord_trade_cost_factor = -0.25
		peace_score_ratio_transferred_to_overlord = 0.05
		lend_lease_tension_with_overlord = -0.1
		faction_influence_contribution_factor = -0.1
	}
	
	ai_subject_wants_higher = {
		factor = 0.0
		modifier = {
			original_tag = CAN
			OR = {
				has_completed_focus = CAN_burn_the_royal_portraits
				has_completed_focus = CAN_swastika_clubs
			}
			add = 10
		}
		modifier = {
			IRQ_PER_should_seek_independence = yes
			add = 10
		}
	}
	
	ai_overlord_wants_lower = {
		factor = 1.0
	}

	ai_overlord_wants_garrison = {
		always = no
	}

	allowed = {
		has_dlc = "Together for Victory"

		if = {
			limit = { has_dlc = "Death or Dishonor" }
			OVERLORD = { NOT = { has_government = fascism } }
		}

	}
	
	can_take_level = {
		IF = {
			limit = {
				original_tag = COG
				has_dlc = "Gotterdammerung"			
			}
			has_completed_focus = CONGO_republic_of_congo
			NOT = {
				has_completed_focus = CONGO_dominion_of_congo
			}
		}
	}

	can_lose_level = {
		IF = {
			limit = {
				OVERLORD = { original_tag = BEL } 
				original_tag = COG
				has_dlc = "Gotterdammerung"			
			}
			NOT = {
				has_completed_focus = CONGO_dominion_of_congo
			}
		}
	}
}
