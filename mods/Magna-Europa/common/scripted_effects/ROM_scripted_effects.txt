### calculating the effect of claim strength in Yugoslavia ####
###############################################################

calculate_claimant_effect = {	
	if = {
		limit = {
			check_variable = { claim_strength_@PREV < 2 }
		}
		add_to_variable = { var = claim_strength_@PREV value = 1 }
		custom_effect_tooltip = romania_add_claims_tt
	}
	else_if = {
		limit = {
			NOT = {
				any_country = {
					has_country_flag = invited_to_split_yugo
					PREV = {
						is_claimed_by = PREV
					}
				}
			}
		}
		add_to_variable = { var = claim_strength_@PREV value = 1 }
		custom_effect_tooltip = newline_tt
		add_claim_by = PREV
	}
	if = {
		limit = {
			check_variable = {claim_strength_ROM > claim_strength_HUN }
			check_variable = {claim_strength_ROM > claim_strength_ITA }
			check_variable = {claim_strength_ROM > claim_strength_GER }
		}
		set_state_flag = strongest_claimant_ROM
		clr_state_flag = strongest_claimant_GER
		clr_state_flag = strongest_claimant_ITA
		clr_state_flag = strongest_claimant_HUN
	}
	if = {
		limit = {
			check_variable = {claim_strength_GER > claim_strength_HUN }
			check_variable = {claim_strength_GER > claim_strength_ITA }
			check_variable = {claim_strength_GER > claim_strength_ROM }
		}
		set_state_flag = strongest_claimant_GER
		clr_state_flag = strongest_claimant_ROM
		clr_state_flag = strongest_claimant_ITA
		clr_state_flag = strongest_claimant_HUN
	}
	if = {
		limit = {
			check_variable = {claim_strength_ITA > claim_strength_HUN }
			check_variable = {claim_strength_ITA > claim_strength_ROM }
			check_variable = {claim_strength_ITA > claim_strength_GER }
		}
		set_state_flag = strongest_claimant_ITA
		clr_state_flag = strongest_claimant_GER
		clr_state_flag = strongest_claimant_ROM
		clr_state_flag = strongest_claimant_HUN
	}
	if = {
		limit = {
			check_variable = {claim_strength_HUN > claim_strength_ROM }
			check_variable = {claim_strength_HUN > claim_strength_ITA }
			check_variable = {claim_strength_HUN > claim_strength_GER }
		}
		set_state_flag = strongest_claimant_HUN
		clr_state_flag = strongest_claimant_GER
		clr_state_flag = strongest_claimant_ITA
		clr_state_flag = strongest_claimant_ROM
	}
}

ROM_update_foreign_relations_dynamic_modifier = {			#ran by on-action if any relation changed
	ROM_clear_diplo_variables = yes
	for_each_scope_loop = {
		array = global.rom_diplomacy_partners
		if = {
			limit = {
				check_variable = { ROM_relation_var > 4.5 }
			}
			ROM_add_country_tier_5_buff = yes
		}
		else_if = {
			limit = {
				check_variable = { ROM_relation_var > 3.5 }
			}
			ROM_add_country_tier_4_buff = yes
		}
		else_if = {
			limit = {
				check_variable = { ROM_relation_var > 2.5 }
			}
			#ROM_add_country_tier_3_buff = yes
		}
		else_if = {
			limit = {
				check_variable = { ROM_relation_var > 1.5 }
			}
			ROM_add_country_tier_2_buff = yes
		}
		if = {	#tier 3 bonus stacks with the stats of 4 OR 5
			limit = {
				check_variable = { ROM_relation_var > 2.5 }
			}
			ROM_add_country_tier_3_buff = yes
		}
	}
	force_update_dynamic_modifier = yes
}

ROM_recalculate_foreign_relations_political_power_cost = {
	set_variable = {
		ROM_international_relations_political_power_cost = 0
	}
	for_each_scope_loop = {
		array = global.rom_diplomacy_partners
		if = {
			limit = {
				has_country_flag = ROM_improving_relations
			}
			ROM = {
				add_to_variable = {
					ROM_international_relations_political_power_cost = ROM_one_country_relations_boosting_cost
				}
			}
		}
		if = {
			limit = {
				has_country_flag = ROM_maintaining_relations_high
			}
			ROM = {
				add_to_variable = {
					ROM_international_relations_political_power_cost = ROM_one_country_relations_boosting_cost
				}
			}
		}
	}
}

		subtract_from_variable = {
			ROM.ROM_international_relations_political_power_cost = ROM.ROM_one_country_relations_boosting_cost
		}
	}
	else = {
		subtract_from_variable = {
			ROM.ROM_international_relations_political_power_cost = ROM.ROM_one_country_relations_maintain_cost
		}
	}
	clr_country_flag = ROM_improving_relations
	clr_country_flag = ROM_maintaining_relations_high
	subtract_from_variable = {
		ROM.ROM_current_improving_countries = 1
	}
