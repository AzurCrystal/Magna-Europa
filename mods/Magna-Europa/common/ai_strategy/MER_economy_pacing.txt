# Magna Europa: Reforged — Base-Mod AI Economy & Production Pacing
# Goals:
#   1. Smooth civilian->military construction rhythm without touching historical focus order or tech queues.
#   2. Stock-gated logistics/production floors so majors neither starve key lines (fighters/tanks/support)
#      nor over-produce trains/trucks/rifles, and minors are never broken by unconditional floors.
#   3. All blocks yield cleanly if a future standalone EAI submod sets global flag `MER_EAI_enabled`.

###############################################################################
# 1. GLOBAL STOCK-GATED LOGISTICS & SURPLUS REBALANCING
###############################################################################

MER_global_supply_trains_floor = {
	enable = {
		NOT = { has_global_flag = MER_EAI_enabled }
		NOT = { original_tag = MAP }
		num_of_military_factories > 14
		has_equipment = { train_equipment < 150 }
	}
	abort_when_not_enabled = yes

	ai_strategy = {
		type = equipment_production_min_factories_archetype
		id = train_equipment
		value = 1
	}
}

# Stop majors from pouring 45-55% of MILs into infinite rifles once a healthy 20k surplus exists;
# automatically turns back off the moment wartime losses dip rifles below 20k.
MER_major_rifle_surplus_rebalance = {
	allowed = {
		OR = {
			original_tag = GER
			original_tag = SOV
			original_tag = ENG
			original_tag = FRA
			original_tag = ITA
		}
	}
	enable = {
		NOT = { has_global_flag = MER_EAI_enabled }
		num_of_military_factories > 40
		has_equipment = { infantry_equipment > 20000 }
	}
	abort_when_not_enabled = yes

	ai_strategy = {
		type = equipment_production_factor
		id = infantry
		value = -30
	}
	ai_strategy = {
		type = equipment_production_factor
		id = fighter
		value = 25
	}
	ai_strategy = {
		type = equipment_production_factor
		id = cas
		value = 15
	}
}
