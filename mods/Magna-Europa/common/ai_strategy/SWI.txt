SWI_historical_placate_GER = {
	allowed = {
		SWI_bba = yes 
	}

	enable = {
		is_historical_focus_on = yes  
	}

	abort = {
		OR = {
			date > 1945.1.1
			NOT = { country_exists = GER }
			has_war_with = GER
			GER = {
				OR = {
					#NOT = {	has_government = fascism }
					has_opinion	= {
						target = ROOT
						value > 80
					}
					AND = {
						has_opinion_modifier = SWI_light_sentence_for_frankfurter
						has_opinion	= {
							target = ROOT
							value > 50
						}
					}
					has_capitulated = yes
					is_subject = yes
					AND = {
						is_in_faction = yes
						is_faction_leader = no
					}
					SWI_is_country_to_balance = no 
				}
			}
		}
	}

	ai_strategy = {
		type = diplo_action_desire
		id = GER
		target = improve_relation
		value = 200
	}
}
