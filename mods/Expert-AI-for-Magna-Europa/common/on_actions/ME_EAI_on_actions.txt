############################################################################################################
# Expert AI for Magna Europa: Reforged — On-Actions
############################################################################################################

on_actions = {
	on_startup = {
		effect = {
			random_country = {
				ME_EAI_startup = yes
			}
		}
	}
	on_daily = {
		effect = {
			ME_EAI_daily_dispatch = yes
		}
	}
	on_weekly = {
		effect = {
			ME_EAI_weekly_dispatch = yes
		}
	}
	on_weekly_SOV = {
		effect = {
			# Zero-cheat pro-player Great Purge Paranoia discipline:
			# Runs right after 07_nsb_on_actions.txt ticks weekly paranoia, keeping SOV_paranoia < 25 so
			# Red Army & Red Air Force suffer ZERO random purges (no dissolved air bureaus, no banned military thinking!).
			# Priority 1: Forge Production Reports (pays exact -25 PP, runs exact vanilla decision w/ +2% CG for 90d & 270d CD).
			# Priority 2: Fallback "Inspection in the Navy" (pays exact -50 PP, runs exact vanilla Navy purge w/ 187d CD,
			#             sacrificing Navy to protect Red Army & Red Air Force when Forge is on cooldown!).
			if = {
				limit = {
					is_ai = yes
					has_country_flag = SOV_paranoia_system_active_flag
					NOT = { has_country_flag = SOV_civil_war_triggered_flag }
					SOV_paranoia_is_finishing_purge_focus = no
				}
				if = {
					limit = {
						check_variable = { SOV_paranoia > 20 }
						has_political_power > 24
						NOT = { has_decision = SOV_forge_production_reports }
						NOT = { has_country_flag = me_eai_sov_forge_cd }
					}
					add_political_power = -25
					set_country_flag = { flag = me_eai_sov_forge_cd value = 1 days = 270 }
					activate_decision = SOV_forge_production_reports
					log = "[GetDateText] | ME-EAI | SOV Paranoia Guard: Forge Production Reports activated (paranoia=[?SOV_paranoia])"
				}
				else_if = {
					limit = {
						check_variable = { SOV_paranoia > 22 }
						has_political_power > 49
						NOT = { has_decision = SOV_inspection_in_the_navy }
						NOT = { has_country_flag = SOV_paranoia_conducting_inspection_flag }
						NOT = { has_country_flag = me_eai_sov_navy_cd }
						NOT = { has_country_flag = SOV_paranoia_random_purge_cooldown_from_focus_flag }
						NOT = { has_country_flag = SOV_paranoia_random_purge_cooldown_from_decision_flag }
						OR = {
							any_navy_leader = { SOV_character_can_be_purged = yes }
							NOT = { has_idea = SOV_anti_soviet_military_thinking_banned_navy }
							NOT = { has_idea = SOV_shipyard_staff_purged }
						}
					}
					add_political_power = -50
					set_country_flag = { flag = me_eai_sov_navy_cd value = 1 days = 187 }
					activate_decision = SOV_inspection_in_the_navy
					log = "[GetDateText] | ME-EAI | SOV Paranoia Guard: Navy Inspection activated (paranoia=[?SOV_paranoia])"
				}
			}
		}
	}
	on_monthly = {
		effect = {
			ME_EAI_on_month_tick = yes
		}
	}
	on_declare_war = {
		effect = {
			ROOT = {
				set_country_flag = { flag = ME_EAI_just_declared_war value = 1 days = 20 }
				ME_EAI_on_war_crisis = yes
			}
			FROM = { ME_EAI_on_war_crisis = yes }
		}
	}
	on_capitulation = {
		effect = {
			# Zero-cheat pro-player timing: when France capitulates to AI Germany under Economy of Conquest,
			# execute vanilla GER_seize_gold_reserves across ALL German-occupied European French cores
			# BEFORE france.14 partitions Vichy the next day. Pays the exact -25 PP cost, applies exact
			# -5 compliance / +10 resistance / 90d overburdened modifier, and marks GER_seize_gold_reserves_state_flag
			# on every state so vanilla's decision can NEVER double-dip France.
			if = {
				limit = {
					ROOT = { original_tag = FRA }
					FROM = {
						original_tag = GER
						is_ai = yes
						has_dynamic_modifier = { modifier = GER_economy_of_conquest_modifier }
						has_political_power > 24
						check_variable = { GER_industrial_consumer_goods_factor > 0.25 }
					}
					ROOT = {
						any_core_state = {
							is_on_continent = europe
							is_controlled_by = GER
							NOT = { has_state_flag = GER_seize_gold_reserves_state_flag }
						}
					}
				}
				GER = {
					add_political_power = -25
					set_temp_variable = { occupied_state_num_factory = 0 }
					add_to_temp_variable = { occupied_state_num_factory = ROOT.num_of_military_factories_in_cores@GER }
					add_to_temp_variable = { occupied_state_num_factory = ROOT.num_of_civilian_factories_in_cores@GER }
					add_to_temp_variable = { occupied_state_num_factory = ROOT.num_of_naval_factories_in_cores@GER }
					ROOT = {
						every_core_state = {
							limit = {
								is_on_continent = europe
								is_controlled_by = GER
								NOT = { has_state_flag = GER_seize_gold_reserves_state_flag }
							}
							add_compliance = -5
							add_resistance = 10
							set_state_flag = GER_seize_gold_reserves_state_flag
							add_dynamic_modifier = { modifier = GER_overburdened_state_modifier days = 90 }
						}
					}
					divide_temp_variable = { occupied_state_num_factory = 200 }
					multiply_temp_variable = { occupied_state_num_factory = -1 }
					add_to_variable = { GER_industrial_consumer_goods_factor = occupied_state_num_factory }
					clamp_variable = { var = GER_industrial_consumer_goods_factor min = 0 }
					GER_reduce_mefo_bills_penalty = yes
				}
			}
			FROM = { ME_EAI_on_war_crisis = yes }
		}
	}
}
