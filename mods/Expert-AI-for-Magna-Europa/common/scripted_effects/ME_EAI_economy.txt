# Expert AI for Magna Europa: Reforged — Phase 2 Economy, Deficit Loop, Resource & Lend-Lease Brain
# Runs inside:
#   - ME_EAI_weekly_tier1_pulse (~15 Tier 1 nations, once/week)
#   - ME_EAI_monthly_brain (staggered 28-day slot, 1-2 nations/day)
#   - ME_EAI_on_war_crisis (instant event-driven hook)
# Uses Schmitt-trigger hysteresis (start < LOW, clear > HIGH) to preserve MIL line production efficiency.
# Exports ONLY O(1) country flags consumed by ME_EAI_economy_strategies.txt & ME_EAI_diplomacy_trade.txt.
############################################################################################################

ME_EAI_update_deficit_and_resource_flags = {
	#-------------------------------------------------------------------------------------------------------
	# 1. Granular Land Equipment Deficit Flags (with Hysteresis -> keeps MIL efficiency 85%+)
	#-------------------------------------------------------------------------------------------------------
	set_temp_variable = { _e_inf_str = 1 }
	if = {
		limit = { check_variable = { num_target_equipment_in_armies_k@infantry_equipment > 5 } }
		set_temp_variable = { _e_inf_str = num_equipment_in_armies_k@infantry_equipment }
		divide_temp_variable = { _e_inf_str = num_target_equipment_in_armies_k@infantry_equipment }
	}

	# Rifles deficit (triggers on genuine field shortfall OR thin Tier-1 reserve; clears only after healthy buffer)
	if = {
		limit = { NOT = { has_country_flag = ME_EAI_def_rifles } }
		if = {
			limit = {
				OR = {
					check_variable = { _e_inf_str < 0.92 }
					AND = {
						has_country_flag = ME_EAI_tier_1
						has_equipment = { infantry_equipment < 3500 }
					}
					AND = {
						has_country_flag = ME_EAI_tier_2
						has_equipment = { infantry_equipment < 1000 }
					}
				}
			}
			set_country_flag = ME_EAI_def_rifles
		}
	}
	else = {
		if = {
			limit = {
				check_variable = { _e_inf_str > 0.96 }
				OR = {
					AND = {
						has_country_flag = ME_EAI_tier_1
						has_equipment = { infantry_equipment > 12000 }
					}
					AND = {
						NOT = { has_country_flag = ME_EAI_tier_1 }
						has_equipment = { infantry_equipment > 3500 }
					}
				}
			}
			clr_country_flag = ME_EAI_def_rifles
		}
	}

	# Superpower Mass-Army Rifle Floor (prevents 400-div Wehrmacht/Red Army from bleeding -31k rifles when armor/air ramp!)
	if = {
		limit = {
			has_country_flag = ME_EAI_def_rifles
			num_divisions > 220
			num_of_military_factories > 79
			OR = {
				check_variable = { _e_inf_str < 0.95 }
				has_equipment = { infantry_equipment < 5000 }
			}
		}
		set_country_flag = ME_EAI_def_rifles_heavy
	}
	else = {
		clr_country_flag = ME_EAI_def_rifles_heavy
	}

	# Acute Superpower Rifle Crisis Surge (stacks to 54 guaranteed rifle MILs so 400+ div armies NEVER lock in ME_EAI_rifles_starved!)
	if = {
		limit = { NOT = { has_country_flag = ME_EAI_def_rifles_crisis } }
		if = {
			limit = {
				has_country_flag = ME_EAI_def_rifles_heavy
				num_of_military_factories > 119
				OR = {
					check_variable = { _e_inf_str < 0.88 }
					has_equipment = { infantry_equipment < 1000 }
				}
			}
			set_country_flag = ME_EAI_def_rifles_crisis
		}
	}
	else = {
		if = {
			limit = {
				OR = {
					NOT = { has_country_flag = ME_EAI_def_rifles_heavy }
					AND = {
						check_variable = { _e_inf_str > 0.93 }
						has_equipment = { infantry_equipment > 4000 }
					}
				}
			}
			clr_country_flag = ME_EAI_def_rifles_crisis
		}
	}

	# Artillery deficit (Tier 1 & industrialized Tier 2; start < 500, stop > 1600)
	if = {
		limit = { NOT = { has_country_flag = ME_EAI_def_arty } }
		if = {
			limit = {
				num_of_military_factories > 24
				has_tech = interwar_artillery
				has_equipment = { artillery_equipment < 500 }
			}
			set_country_flag = ME_EAI_def_arty
		}
	}
	else = {
		if = {
			limit = {
				OR = {
					num_of_military_factories < 20
					has_equipment = { artillery_equipment > 1600 }
				}
			}
			clr_country_flag = ME_EAI_def_arty
		}
	}

	# Anti-Tank deficit (start < 350, stop > 1100)
	if = {
		limit = { NOT = { has_country_flag = ME_EAI_def_at } }
		if = {
			limit = {
				has_country_flag = ME_EAI_tier_1
				num_of_military_factories > 29
				has_tech = interwar_antitank
				has_equipment = { anti_tank_equipment < 350 }
			}
			set_country_flag = ME_EAI_def_at
		}
	}
	else = {
		if = {
			limit = {
				OR = {
					num_of_military_factories < 24
					has_equipment = { anti_tank_equipment > 1100 }
				}
			}
			clr_country_flag = ME_EAI_def_at
		}
	}

	# Medium Tank stockpile floor for industrialized armored powers (balanced so GER keeps 45-55 Panzers WITHOUT starving fighters/rifles!)
	if = {
		limit = { NOT = { has_country_flag = ME_EAI_def_med_tank } }
		if = {
			limit = {
				has_country_flag = ME_EAI_want_armor
				has_tech = basic_medium_tank_chassis
				num_of_military_factories > 59
				OR = {
					has_equipment = { medium_tank_chassis < 400 }
					AND = {
						OR = { original_tag = GER original_tag = SOV }
						has_equipment = { medium_tank_chassis < 650 }
					}
				}
			}
			set_country_flag = ME_EAI_def_med_tank
		}
	}
	else = {
		if = {
			limit = {
				OR = {
					NOT = { has_country_flag = ME_EAI_want_armor }
					num_of_military_factories < 50
					AND = {
						NOT = { original_tag = GER original_tag = SOV }
						has_equipment = { medium_tank_chassis > 1100 }
					}
					AND = {
						OR = { original_tag = GER original_tag = SOV }
						has_equipment = { medium_tank_chassis > 1500 }
					}
				}
			}
			clr_country_flag = ME_EAI_def_med_tank
		}
	}

	# Medium Flame Tank support floor (start < 120, stop > 360)
	if = {
		limit = { NOT = { has_country_flag = ME_EAI_def_flame_tank } }
		if = {
			limit = {
				has_country_flag = ME_EAI_want_armor
				has_tech = basic_medium_tank_chassis
				has_tech = tech_engineers2
				num_of_military_factories > 59
				has_equipment = { medium_tank_flame_chassis < 120 }
			}
			set_country_flag = ME_EAI_def_flame_tank
		}
	}
	else = {
		if = {
			limit = {
				OR = {
					NOT = { has_country_flag = ME_EAI_want_armor }
					num_of_military_factories < 50
					has_equipment = { medium_tank_flame_chassis > 360 }
				}
			}
			clr_country_flag = ME_EAI_def_flame_tank
		}
	}

	#-------------------------------------------------------------------------------------------------------
	# 2. Air Superiority & CAS Deficit Flags (Green-Air-First, Scaled by Industrial Capacity)
	#-------------------------------------------------------------------------------------------------------
	clr_country_flag = ME_EAI_def_fighters
	clr_country_flag = ME_EAI_def_fighters_heavy
	clr_country_flag = ME_EAI_def_cas

	if = {
		limit = {
			has_country_flag = ME_EAI_tier_1
			num_of_military_factories > 39
			OR = {
				original_tag = GER
				original_tag = SOV
				original_tag = ENG
				original_tag = USA
				original_tag = ITA
				original_tag = FRA
			}
		}
		# Superpower wartime fighter ramp (>= 100 MILs, 1940+, target >= 3000 deployed fighters)
		if = {
			limit = {
				num_of_military_factories > 99
				has_global_flag = ME_EAI_era_1940
				OR = { original_tag = GER original_tag = SOV original_tag = ENG original_tag = USA }
				has_deployed_air_force_size = { type = fighter size < 3000 }
			}
			set_country_flag = ME_EAI_def_fighters
			set_country_flag = ME_EAI_def_fighters_heavy
		}
		# All 6 majors baseline fighter floor (target >= 1200 deployed fighters)
		else_if = {
			limit = {
				has_deployed_air_force_size = { type = fighter size < 1200 }
			}
			set_country_flag = ME_EAI_def_fighters
		}
		# Once fighter wing strength is healthy, push CAS up to >= 800 deployed aircraft
		else_if = {
			limit = {
				num_of_military_factories > 59
				has_deployed_air_force_size = { type = cas size < 800 }
			}
			set_country_flag = ME_EAI_def_cas
		}
	}

	#-------------------------------------------------------------------------------------------------------
	# 3. Convoy Lifeline Floor (Dockyard-driven with Hysteresis for Maritime & Amphibious Majors)
	#-------------------------------------------------------------------------------------------------------
	if = {
		limit = { NOT = { has_country_flag = ME_EAI_def_convoys } }
		if = {
			limit = {
				OR = {
					AND = {
						OR = { original_tag = ENG original_tag = USA }
						has_equipment = { convoy < 450 }
					}
					AND = {
						OR = { original_tag = ITA original_tag = GER original_tag = FRA }
						has_equipment = { convoy < 200 }
					}
				}
			}
			set_country_flag = ME_EAI_def_convoys
		}
	}
	else = {
		if = {
			limit = {
				OR = {
					AND = {
						OR = { original_tag = ENG original_tag = USA }
						has_equipment = { convoy > 750 }
					}
					AND = {
						OR = { original_tag = ITA original_tag = GER original_tag = FRA }
						has_equipment = { convoy > 380 }
					}
					NOT = {
						OR = { original_tag = ENG original_tag = USA original_tag = ITA original_tag = GER original_tag = FRA }
					}
				}
			}
			clr_country_flag = ME_EAI_def_convoys
		}
	}

	#-------------------------------------------------------------------------------------------------------
	# 4. Resource Shortage & Fuel Pressure (Triggers Honest CIV-Paid Synthetic Refineries & Silos)
	#-------------------------------------------------------------------------------------------------------
	if = {
		limit = {
			has_country_flag = ME_EAI_tier_1
			has_tech = synth_oil_experiments
			num_of_civilian_factories > 29
			OR = {
				check_variable = { resource@rubber < -10 }
				AND = {
					has_country_flag = ME_EAI_axis_synth
					check_variable = { resource_imported@rubber > 12 }
				}
				AND = {
					original_tag = SOV
					date > 1939.6.1
					num_of_civilian_factories > 80
					check_variable = { resource@rubber < 10 }
				}
				AND = {
					has_country_flag = ME_EAI_axis_synth
					NOT = { original_tag = SOV }
					has_war = yes
					fuel_ratio < 0.55
				}
			}
		}
		set_country_flag = ME_EAI_need_refineries
	}
	else = {
		clr_country_flag = ME_EAI_need_refineries
	}

	if = {
		limit = {
			has_country_flag = ME_EAI_axis_synth
			NOT = { original_tag = SOV }
			has_war = no
			date > 1938.1.1
			fuel_ratio > 0.85
			num_of_civilian_factories > 45
		}
		set_country_flag = ME_EAI_need_silos
	}
	else = {
		clr_country_flag = ME_EAI_need_silos
	}

	#-------------------------------------------------------------------------------------------------------
	# 5. Expanded Honest Law PP Guard (Single Owner of ME_EAI_need_law_pp: Conscription + War/Total Economy)
	#-------------------------------------------------------------------------------------------------------
	if = {
		limit = {
			has_war = yes
			OR = {
				# Urgent manpower law upgrade needed
				AND = {
					NOT = { has_idea = scraping_the_barrel }
					OR = {
						has_manpower < 300000
						AND = {
							OR = { original_tag = GER original_tag = SOV }
							has_manpower < 850000
							OR = {
								has_idea = volunteer_only
								has_idea = limited_conscription
								has_idea = extensive_conscription
							}
						}
						AND = {
							original_tag = GER
							has_idea = totaler_krieg_economy
							has_manpower < 800000
						}
					}
				}
				# Urgent war economy upgrade needed (prevents ROM/HUN/FIN staying on civilian/partial economy in wartime!)
				AND = {
					NOT = { original_tag = USA }
					has_war_support > 0.49
					OR = {
						has_idea = civilian_economy
						has_idea = low_economic_mobilisation
						has_idea = partial_economic_mobilisation
					}
				}
				# Soviet Total Mobilization push once wartime Propaganda lifts War Support >= 80%!
				AND = {
					original_tag = SOV
					has_war_with = GER
					has_war_support > 0.79
					has_idea = war_economy
				}
			}
		}
		set_country_flag = ME_EAI_need_law_pp
	}
	else = {
		clr_country_flag = ME_EAI_need_law_pp
	}

	#-------------------------------------------------------------------------------------------------------
	# 6. German Historical Four-Year Plan MEFO / Economy of Conquest CG Guard (O(1) Flag)
	#    Ensures GER always prioritizes Price Control & Gold Seizure decisions whenever CG > 25%,
	#    preventing the 1942 100%-CG bankruptcy death spiral on the historical 4YP path.
	#-------------------------------------------------------------------------------------------------------
	if = {
		limit = {
			original_tag = GER
			OR = {
				has_dynamic_modifier = { modifier = GER_mefo_bills_modifier }
				has_dynamic_modifier = { modifier = GER_economy_of_conquest_modifier }
			}
			check_variable = { GER_industrial_consumer_goods_factor > 0.25 }
		}
		set_country_flag = ME_EAI_ger_mefo_high
	}
	else = {
		clr_country_flag = ME_EAI_ger_mefo_high
	}

	#-------------------------------------------------------------------------------------------------------
	# 7. Soviet Great Purge PP Reserve, Propaganda, Gosproyektstroy, Urals Relocation & Stavka Reserves (O(1) Flags)
	#-------------------------------------------------------------------------------------------------------
	if = {
		limit = {
			original_tag = SOV
			has_country_flag = SOV_paranoia_system_active_flag
			check_variable = { SOV_paranoia > 12 }
			has_political_power < 55
		}
		set_country_flag = ME_EAI_sov_purge_pp_reserve
	}
	else = {
		clr_country_flag = ME_EAI_sov_purge_pp_reserve
	}

	if = {
		limit = {
			original_tag = SOV
			NOT = { has_country_flag = ME_EAI_sov_purge_pp_reserve }
			OR = {
				has_war_with = GER
				AND = {
					has_completed_focus = SOV_expand_the_agitprop
					NOT = { has_country_flag = SOV_propaganda_center_days_remaining_flag }
				}
				AND = {
					has_completed_focus = SOV_foreign_experts
					has_idea = SOV_gosproyektstroy_1
				}
			}
		}
		set_country_flag = ME_EAI_sov_pp_decisions
	}
	else = {
		clr_country_flag = ME_EAI_sov_pp_decisions
	}

	if = {
		limit = {
			original_tag = SOV
			has_war_with = GER
			surrender_progress > 0.03
			num_divisions < 580
			has_manpower > 500000
		}
		set_country_flag = ME_EAI_sov_stavka_reserves
	}
	else = {
		clr_country_flag = ME_EAI_sov_stavka_reserves
	}
}

#-----------------------------------------------------------------------------------------------------------
# ME_EAI_update_lend_lease_flags
#   Computes precision wartime coalition lend-lease desire flags once per month per active donor.
#-----------------------------------------------------------------------------------------------------------
ME_EAI_update_lend_lease_flags = {
	clr_country_flag = ME_EAI_ll_to_SOV
	clr_country_flag = ME_EAI_ll_to_ENG
	clr_country_flag = ME_EAI_ll_to_FRA
	clr_country_flag = ME_EAI_ll_to_ROM
	clr_country_flag = ME_EAI_ll_to_HUN
	clr_country_flag = ME_EAI_ll_to_FIN
	clr_country_flag = ME_EAI_ll_to_ITA

	# USA -> ENG / SOV / FRA (Arsenal of Democracy)
	if = {
		limit = {
			original_tag = USA
			num_of_military_factories > 80
			NOT = { has_country_flag = ME_EAI_rifles_starved }
			OR = {
				has_war = yes
				has_completed_focus = USA_lend_lease_act
				threat > 0.70
			}
		}
		if = {
			limit = {
				ENG = {
					exists = yes
					has_war_with = GER
					NOT = { has_war_with = PREV }
				}
			}
			set_country_flag = ME_EAI_ll_to_ENG
		}
		if = {
			limit = {
				SOV = {
					exists = yes
					has_war_with = GER
					NOT = { has_war_with = PREV }
				}
			}
			set_country_flag = ME_EAI_ll_to_SOV
		}
		if = {
			limit = {
				FRA = {
					exists = yes
					has_capitulated = no
					has_war_with = GER
					NOT = { has_war_with = PREV }
				}
			}
			set_country_flag = ME_EAI_ll_to_FRA
		}
	}

	# ENG -> SOV (Arctic Convoys once Britain's own rifles are healthy)
	if = {
		limit = {
			original_tag = ENG
			has_war_with = GER
			NOT = { has_country_flag = ME_EAI_rifles_starved }
			NOT = { has_country_flag = ME_EAI_def_rifles }
			SOV = {
				exists = yes
				has_war_with = GER
				NOT = { has_war_with = PREV }
			}
		}
		set_country_flag = ME_EAI_ll_to_SOV
	}

	# GER -> Axis Partners (when GER has rifle surplus and partners fight SOV/ENG)
	if = {
		limit = {
			original_tag = GER
			has_war = yes
			NOT = { has_country_flag = ME_EAI_rifles_starved }
			has_equipment = { infantry_equipment > 15000 }
		}
		if = {
			limit = {
				ROM = { exists = yes is_in_faction_with = GER has_war = yes has_equipment = { infantry_equipment < 8000 } }
			}
			set_country_flag = ME_EAI_ll_to_ROM
		}
		if = {
			limit = {
				HUN = { exists = yes is_in_faction_with = GER has_war = yes has_equipment = { infantry_equipment < 8000 } }
			}
			set_country_flag = ME_EAI_ll_to_HUN
		}
		if = {
			limit = {
				FIN = { exists = yes has_war_with = SOV NOT = { has_war_with = GER } has_equipment = { infantry_equipment < 8000 } }
			}
			set_country_flag = ME_EAI_ll_to_FIN
		}
		if = {
			limit = {
				ITA = { exists = yes is_in_faction_with = GER has_war = yes has_equipment = { infantry_equipment < 10000 } }
			}
			set_country_flag = ME_EAI_ll_to_ITA
		}
	}
}
