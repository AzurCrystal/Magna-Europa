#-----------------------------------------------------------------------------------------------------------
# ME_EAI_events.1 — Day-1 Welcome, Expert AI 5.0 Tribute & Quick Difficulty Selection
#-----------------------------------------------------------------------------------------------------------
country_event = {
	id = ME_EAI_events.1
	title = ME_EAI_events.1.t
	desc = ME_EAI_events.1.desc
	picture = GFX_report_event_generic_read_write

	is_triggered_only = yes

	# Option A: Keep Game Rule / Standard (Zero-Cheat — Recommended)
	option = {
		name = ME_EAI_events.1.a
		ai_chance = { base = 100 }
		if = {
			limit = { has_global_flag = ME_EAI_diff_custom }
			country_event = { id = ME_EAI_events.2 }
		}
		else = {
			custom_effect_tooltip = ME_EAI_events.1.a.tt
		}
	}

	# Option B: Switch immediately to Challenging (+15% AI Industry & Supply)
	option = {
		name = ME_EAI_events.1.b
		ai_chance = { base = 0 }
		clr_global_flag = ME_EAI_diff_standard
		clr_global_flag = ME_EAI_diff_brutal
		set_global_flag = ME_EAI_diff_challenging
		ME_EAI_sync_all_difficulty_ideas = yes
		custom_effect_tooltip = ME_EAI_events.1.b.tt
	}

	# Option C: Switch immediately to Brutal (+30% AI Industry & Combat)
	option = {
		name = ME_EAI_events.1.c
		ai_chance = { base = 0 }
		clr_global_flag = ME_EAI_diff_standard
		clr_global_flag = ME_EAI_diff_challenging
		set_global_flag = ME_EAI_diff_brutal
		ME_EAI_sync_all_difficulty_ideas = yes
		custom_effect_tooltip = ME_EAI_events.1.c.tt
	}

	# Option D: Open Bloc-Focus Boost Menu (e.g. boost only Axis / USSR / Allies / Anti-Player)
	option = {
		name = ME_EAI_events.1.d
		ai_chance = { base = 0 }
		country_event = { id = ME_EAI_events.3 }
	}
}
